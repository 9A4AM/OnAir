cmake_minimum_required(VERSION 3.20)
if(APPLE)
  project(OnAir LANGUAGES C CXX OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
else()
  project(OnAir LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_CXX_FLAGS_RELEASE "-O3")
  set(CMAKE_OBJCXX_FLAGS_RELEASE "-O3")
endif()

option(DECT2_BUILD_APP "Build the GUI application" ${APPLE})
set(DECT2_UI_BACKEND "" CACHE STRING "Graphics back end of the GUI: Metal (default on macOS) or OpenGL3 (default elsewhere)")
if(NOT DECT2_UI_BACKEND)
  if(APPLE)
    set(DECT2_UI_BACKEND Metal)
  else()
    set(DECT2_UI_BACKEND OpenGL3)
  endif()
endif()
option(DECT2_BUILD_TESTS "Build the tests, tools and benchmarks" ON)
option(DECT2_PORTABLE "Use the portable DSP kernels instead of Accelerate (macOS)" OFF)
option(DECT2_WITH_SOAPY "Support generic radios (Airspy, SDRplay, RTL-SDR, PlutoSDR, LimeSDR, ...) through SoapySDR when it is installed" ON)
option(DECT2_NO_SIMD "Disable the SIMD intrinsics and use the plain C++ code" OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(HACKRF REQUIRED libhackrf)   # plain variables: some package managers list include paths that do not exist
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
if(DECT2_BUILD_APP)
  pkg_check_modules(GLFW REQUIRED IMPORTED_TARGET glfw3)
endif()

if(DECT2_WITH_SOAPY)
  pkg_check_modules(SOAPY SoapySDR)
endif()

add_library(dect2core STATIC core/src/source.cpp core/src/spectrum.cpp core/src/engine.cpp core/src/t2_tables.cpp core/src/t2gen.cpp core/src/t2rx.cpp core/src/t2pilots.cpp core/src/t2ofdm.cpp core/src/ldpc.cpp core/src/t2l1.cpp core/src/t2interleave.cpp core/src/t2fec_tables.cpp core/src/t2fec.cpp core/src/t2plp.cpp core/src/resampler.cpp core/src/bbunpack.cpp core/src/ts.cpp core/src/tsout.cpp core/src/scanner.cpp core/src/gain.cpp core/src/channel.cpp core/src/direction.cpp core/src/dsp_compat.cpp core/src/bandwidth.cpp core/src/atsc.cpp core/src/atsc_gen.cpp core/src/atsc_rx.cpp core/src/quality.cpp core/src/teletext.cpp core/src/dvbt.cpp core/src/dvbt_fec.cpp core/src/dvbt_gen.cpp core/src/dvbt_rx.cpp core/src/player.cpp core/src/audioout.cpp core/src/t2_pilot_tables.cpp)
target_include_directories(dect2core PUBLIC core/include)
if(SOAPY_FOUND)
  message(STATUS "SoapySDR ${SOAPY_VERSION}: generic radios enabled")
  target_sources(dect2core PRIVATE core/src/source_soapy.cpp)
  target_compile_definitions(dect2core PUBLIC DECT2_HAVE_SOAPY)
  target_include_directories(dect2core PRIVATE ${SOAPY_INCLUDE_DIRS})
  target_link_directories(dect2core PUBLIC ${SOAPY_LIBRARY_DIRS})
  target_link_libraries(dect2core PUBLIC ${SOAPY_LIBRARIES})
else()
  message(STATUS "SoapySDR not found: only the HackRF is supported (install libsoapysdr-dev / soapysdr to add other radios)")
endif()
# plain complex multiplication without the NaN/Inf fix-up calls (a large speed-up for std::complex<float>); the spelling depends on the compiler
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag(-fcx-limited-range DECT2_HAS_CX_LIMITED)
check_cxx_compiler_flag(-fcomplex-arithmetic=basic DECT2_HAS_COMPLEX_BASIC)
if(DECT2_HAS_CX_LIMITED)
  target_compile_options(dect2core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fcx-limited-range>)
elseif(DECT2_HAS_COMPLEX_BASIC)
  target_compile_options(dect2core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fcomplex-arithmetic=basic>)
endif()
if(DECT2_PORTABLE)
  target_compile_definitions(dect2core PUBLIC DECT2_PORTABLE)
endif()
if(DECT2_NO_SIMD)
  target_compile_definitions(dect2core PUBLIC DECT2_NO_SIMD)
endif()
target_include_directories(dect2core PUBLIC ${HACKRF_INCLUDE_DIRS})
target_link_directories(dect2core PUBLIC ${HACKRF_LIBRARY_DIRS})
target_link_libraries(dect2core PUBLIC ${HACKRF_LIBRARIES} PkgConfig::FFMPEG)
if(APPLE)
  # Metal LDPC decoder and CoreAudio output
  target_sources(dect2core PRIVATE core/src/gpu_ldpc.mm core/src/audioout_coreaudio.mm)
  set_source_files_properties(core/src/gpu_ldpc.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(dect2core PUBLIC "-framework Accelerate" "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation" "-framework Metal" "-framework Foundation" "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo" pthread iconv)
else()
  # no GPU decoder yet (stub), audio through miniaudio
  target_sources(dect2core PRIVATE core/src/gpu_ldpc_stub.cpp core/src/audioout_miniaudio.cpp)
  target_include_directories(dect2core PRIVATE third_party/miniaudio)
  find_package(Threads REQUIRED)
  target_link_libraries(dect2core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
  if(NOT WIN32)
    target_link_libraries(dect2core PUBLIC m)
  endif()
  find_package(Iconv QUIET)
  if(Iconv_FOUND AND NOT Iconv_IS_BUILT_IN)
    target_link_libraries(dect2core PUBLIC Iconv::Iconv)
  endif()
endif()

if(DECT2_BUILD_APP)
  set(IMGUI_SOURCES
    third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
    third_party/imgui/backends/imgui_impl_glfw.cpp third_party/implot/implot.cpp third_party/implot/implot_items.cpp)
  set(APP_SOURCES app/main.cpp app/gfx_png.cpp)
  if(DECT2_UI_BACKEND STREQUAL "Metal")
    if(NOT APPLE)
      message(FATAL_ERROR "The Metal back end needs macOS")
    endif()
    list(APPEND IMGUI_SOURCES third_party/imgui/backends/imgui_impl_metal.mm)
    list(APPEND APP_SOURCES app/gfx_metal.mm)
  else()
    list(APPEND IMGUI_SOURCES third_party/imgui/backends/imgui_impl_opengl3.cpp)
    list(APPEND APP_SOURCES app/gfx_gl.cpp)
  endif()
  if(APPLE)
    list(APPEND APP_SOURCES app/platform_cocoa.mm)
  else()
    list(APPEND APP_SOURCES app/platform_posix.cpp)
  endif()

  add_library(imgui STATIC ${IMGUI_SOURCES})
  target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends third_party/implot)
  target_link_libraries(imgui PUBLIC PkgConfig::GLFW)
  if(APPLE)
    target_link_libraries(imgui PUBLIC "-framework Cocoa" "-framework IOKit" "-framework CoreVideo" "-framework QuartzCore")
    if(DECT2_UI_BACKEND STREQUAL "Metal")
      target_link_libraries(imgui PUBLIC "-framework Metal" "-framework MetalKit")
    else()
      target_link_libraries(imgui PUBLIC "-framework OpenGL")
      target_compile_definitions(imgui PUBLIC GL_SILENCE_DEPRECATION)
    endif()
    target_compile_options(imgui PRIVATE -fobjc-arc)
  else()
    target_link_libraries(imgui PUBLIC ${CMAKE_DL_LIBS})
  endif()

  add_executable(dect2 ${APP_SOURCES})
  target_include_directories(dect2 PRIVATE app)
  if(APPLE)
    target_compile_options(dect2 PRIVATE -fobjc-arc)
  endif()
  target_link_libraries(dect2 PRIVATE dect2core imgui)
endif()
