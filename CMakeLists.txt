  if(DECT2_CCACHE_PROGRAM)
    set(CMAKE_C_COMPILER_LAUNCHER ${DECT2_CCACHE_PROGRAM})
    set(CMAKE_CXX_COMPILER_LAUNCHER ${DECT2_CCACHE_PROGRAM})
  endif()
endif()

  # Rust SAME / ACARS / ADS-B protocol layers against the C++ ones: a generated copy of each C++ file (namespace ..._ref, cmake/gen_rs_ref.cmake) is
  # the reference of tests/<mode>/test_<mode>_proto_rs.cpp; fuzz_<mode>_proto runs seeded mutations (meant for DECT2_SANITIZE=ON)
  function(onair_rust_diff_tests MOD NAME DIR BASE HDR FROM TO EXTRA INC)
    string(TOUPPER ${MOD} _MU)
    if(NOT ONAIR_RUST_${_MU})
      return()
    endif()
    set(_dir ${CMAKE_BINARY_DIR}/${MOD}_ref)
    add_custom_command(OUTPUT ${_dir}/${NAME}_ref.h ${_dir}/${NAME}_ref.cpp
      COMMAND ${CMAKE_COMMAND} -DSRC=${CMAKE_SOURCE_DIR} -DOUT=${_dir} -DNAME=${NAME} -DHDR=${HDR} -DCPP=${DIR}/${BASE}.cpp -DDEF=DECT2_RUST_${_MU}
              "-DFROM=${FROM}" "-DTO=${TO}" "-DEXTRA=${EXTRA}" -P ${CMAKE_SOURCE_DIR}/cmake/gen_rs_ref.cmake
      DEPENDS core/include/${HDR} core/src/${DIR}/${BASE}.cpp cmake/gen_rs_ref.cmake VERBATIM)
    if(MOD STREQUAL "adsb" OR MOD STREQUAL "p25" OR MOD STREQUAL "tetra")   # same flags as the production file (see the note at dect2core)
      set_source_files_properties(${_dir}/${NAME}_ref.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-fno-builtin-sin;-fno-builtin-cos")
    endif()
    if(MOD STREQUAL "files")   # same flags as the production file (see the note at dect2core)
      set_source_files_properties(${_dir}/${NAME}_ref.cpp core/src/${DIR}/${BASE}.cpp core/src/${DIR}/${BASE}_rs.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-fno-builtin-pow")
    endif()
    set(_srcs core/src/${DIR}/${BASE}.cpp core/src/${DIR}/${BASE}_rs.cpp)
    set_source_files_properties(${_srcs} PROPERTIES COMPILE_DEFINITIONS DECT2_RUST_${_MU})
    add_executable(test_${MOD}_proto_rs tests/${MOD}/test_${MOD}_proto_rs.cpp ${_dir}/${NAME}_ref.cpp ${_srcs})
    target_include_directories(test_${MOD}_proto_rs PRIVATE core/include ${_dir} ${INC})
    target_link_libraries(test_${MOD}_proto_rs PRIVATE onair_rs)
    add_test(NAME ${MOD}_proto_rs COMMAND test_${MOD}_proto_rs)
    add_executable(fuzz_${MOD}_proto tests/${MOD}/fuzz_${MOD}_proto.cpp ${_srcs})
    target_include_directories(fuzz_${MOD}_proto PRIVATE core/include ${INC})
    target_compile_definitions(fuzz_${MOD}_proto PRIVATE DECT2_RUST_${_MU})
    target_link_libraries(fuzz_${MOD}_proto PRIVATE onair_rs)
    add_test(NAME fuzz_${MOD}_proto COMMAND fuzz_${MOD}_proto)
  endfunction()

  function(onair_link_iconv tgt)   # dvb_text.cpp calls iconv: libiconv on macOS, a separate library on some systems, built into glibc elsewhere

  # DVB service information: four layers in one test program (text_charset.cpp, dvb_text.cpp, teletext.cpp, dvbsub.cpp), each with a generated reference copy; the
  # dvb_text copy calls the text_charset copy. The whole decoder classes of teletext.cpp and dvbsub.cpp are in their copies too (end-to-end comparison).
  if(ONAIR_RUST_DVBSI)
    set(_dir ${CMAKE_BINARY_DIR}/dvbsi_ref)
    set(_L_text_charset "dect2/common/text_charset.h|common/text_charset.cpp|namespace dect2::text {|namespace dect2::text_ref {|")
    set(_L_dvb_text "dect2/tv/dvb_text.h|tv/dvb_text.cpp|namespace dect2::dvbsi {|namespace dect2::dvbsi_ref {|#include \"dect2/common/text_charset.h\"=>#include \"text_charset_ref.h\"@@namespace textns = dect2::text=>namespace textns = dect2::text_ref")
    set(_L_teletext "dect2/tv/teletext.h|tv/teletext.cpp|namespace dect2 {|namespace dect2_ref {|")
    set(_L_dvbsub "dect2/tv/dvbsub.h|tv/dvbsub.cpp|namespace dect2 {|namespace dect2_ref {|")
    set(_refsrc "")
    foreach(_n text_charset dvb_text teletext dvbsub)
      string(REPLACE "|" ";" _f "${_L_${_n}}")

                "-DFROM=${_from}" "-DTO=${_to}" "-DEXTRA=" "-DCPP_REPL=${_repl}" -P ${CMAKE_SOURCE_DIR}/cmake/gen_rs_ref.cmake
        DEPENDS core/include/${_h} core/src/${_c} cmake/gen_rs_ref.cmake VERBATIM)

    endif()
  endforeach()

  set(_hdc_fakes "")
  foreach(v 0 1 2 3)
    add_library(fake_faad_hdc${v} MODULE tests/hdr/fake_faad_hdc.c)
    set_target_properties(fake_faad_hdc${v} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
    target_compile_definitions(fake_faad_hdc${v} PRIVATE FAKE_VARIANT=${v})
    list(APPEND _hdc_fakes fake_faad_hdc${v})
  endforeach()

    if(TARGET ${t})
      add_dependencies(${t} ${_hdc_fakes})
      target_compile_definitions(${t} PRIVATE "FAKE_HDC_GOOD=\"$<TARGET_FILE:fake_faad_hdc0>\"" "FAKE_HDC_BADVER=\"$<TARGET_FILE:fake_faad_hdc1>\""
                                 "FAKE_HDC_NOINIT=\"$<TARGET_FILE:fake_faad_hdc2>\"" "FAKE_HDC_BADLAYOUT=\"$<TARGET_FILE:fake_faad_hdc3>\"")

    add_library(fake_onair_dra${v} MODULE tests/cdr/fake_onair_dra.c)

  endforeach()
  foreach(t test_cdr_dra_addon)

      add_dependencies(${t} ${_dra_fakes})
      target_compile_definitions(${t} PRIVATE "FAKE_DRA_GOOD=\"$<TARGET_FILE:fake_onair_dra0>\"" "FAKE_DRA_NOFUNC=\"$<TARGET_FILE:fake_onair_dra1>\""
                                 "FAKE_DRA_NOCREATE=\"$<TARGET_FILE:fake_onair_dra2>\"")
    endif()
  endforeach()
  if(TARGET test_eczas_wav)

  endif()
  if(TARGET test_uat_proto AND TARGET test_uat_gen)
    add_custom_target(test_uat DEPENDS test_uat_proto test_uat_gen)
  endif()

  if(TEST mesh_presets_meshcore)
    set_tests_properties(mesh_presets_meshcore PROPERTIES ENVIRONMENT "ONAIR_TEST_DATA=${CMAKE_SOURCE_DIR}/tests/data")
  endif()

  # Long-running tests (30 s and more): the quick CI on every push skips them (ctest -LE slow),
  # the full CI runs them nightly and on release tags.
  foreach(t gain overload channel t2_payload dvbt_rx dab_rx atsc3_synth_options
            atv_impair atv_interf atv_engine drm_impair drm_rx adsb_engine ism_rx
            gnss_loops gnss_rates gnss_sweep gnss_rate20 gnss_gap gnss_qzss_sbas gnss_galileo gnss_beidou gnss_glonass iq_modes)
    if(TEST ${t})
      set_tests_properties(${t} PROPERTIES LABELS slow)
    endif()
  endforeach()
  # the GNSS tests run for minutes (gnss_beidou ~14 min alone), so give them room under parallel load
  foreach(t gnss_beidou gnss_galileo gnss_glonass gnss_qzss_sbas)
    if(TEST ${t})
      set_tests_properties(${t} PROPERTIES TIMEOUT 2400)
    endif()
  endforeach()
  # the tests measured at 20 s or more (tests/long_tests.txt, regenerated with tools/test_times.sh after a full run)
  file(STRINGS "${CMAKE_SOURCE_DIR}/tests/long_tests.txt" DECT2_LONG_TESTS REGEX "^[A-Za-z0-9_]+$")
  foreach(t ${DECT2_LONG_TESTS})
    if(TEST ${t})
      set_tests_properties(${t} PROPERTIES LABELS slow)
    endif()
  endforeach()
  # every test of a mode with _slow_ in its name
  get_property(DECT2_ALL_TESTS DIRECTORY PROPERTY TESTS)
  foreach(t ${DECT2_ALL_TESTS})
    if(t MATCHES "_slow_")
      set_tests_properties(${t} PROPERTIES LABELS slow)
    endif()
  endforeach()
  # no test plays sound out loud: the audio device still runs, with silence (core/src/audio/audioout.cpp)
  set_property(TEST ${DECT2_ALL_TESTS} APPEND PROPERTY ENVIRONMENT "DECT2_MUTE=1")
endif()
